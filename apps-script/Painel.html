Configuração — Google Sheets via Apps Script
Configure uma vez para salvar o histórico automaticamente.
1Abra script.google.com → "Novo projeto"
2Cole o código abaixo (clique para copiar) e salve com Ctrl+S
3Implantar → Nova implantação → App da Web → Executar como: Eu → Acesso: Qualquer pessoa
4Copie a URL do App da Web e cole abaixo
// Cria/usa uma planilha na SUA conta automaticamente. // Não depende de ID externo — evita erro de permissão. function getSheet_() { var props = PropertiesService.getScriptProperties(); var id = props.getProperty('SHEET_ID'); var ss; if (id) { try { ss = SpreadsheetApp.openById(id); } catch(e) { ss = null; } } if (!ss) { ss = SpreadsheetApp.create('Custos de Obra — Dashboard'); props.setProperty('SHEET_ID', ss.getId()); } var sheet = ss.getSheetByName('Historico') || ss.insertSheet('Historico'); if (sheet.getLastRow() === 0) { sheet.appendRow(['Data/Hora','Obra','Categoria','Item','Orçado (R$)','Realizado (R$)','Desvio (R$)','Venda (R$)','Imposto (R$)','Lucro Liq (R$)','Margem (%)','Status']); sheet.getRange(1,1,1,12).setFontWeight('bold').setBackground('#e8f0fe'); sheet.setFrozenRows(1); } return sheet; } function doPost(e) { try { var raw = (e.parameter && e.parameter.payload) ? e.parameter.payload : (e.postData ? e.postData.contents : null); if (!raw) throw new Error('Payload vazio'); var data = JSON.parse(raw); var sheet = getSheet_(); data.rows.forEach(function(r){ sheet.appendRow(r); }); return ContentService .createTextOutput(JSON.stringify({ok:true, linhas:data.rows.length, sheetUrl:sheet.getParent().getUrl()})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ok:false, erro:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { var sheet = getSheet_(); return ContentService .createTextOutput(JSON.stringify({rows:sheet.getDataRange().getValues(), sheetUrl:sheet.getParent().getUrl()})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({rows:[], erro:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
